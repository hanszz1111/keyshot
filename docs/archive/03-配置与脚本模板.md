## 一、KeyShot headless：多机位 Beauty 出图

```python
# scripts/keyshot_passes.py
# 用法： keyshot_headless -headless "D:/Models/LS-360G.ksp" -script "D:/Dsektop/AI渲染/scripts/keyshot_passes.py"
# 注意：Headless scripting 是 KeyShot Pro 功能（9.3+）

import lux
import os
import json

OUT_ROOT = r"D:\Dsektop\AI渲染\output\passes"
W = H = 2048

ASSETS = [
    # (ksp 路径, SKU, 要输出的机位列表)
    (r"D:\Models\LS-360G.ksp", "LS-360G", ["front", "3q4_left", "3q4_right", "top"]),
]

# 机位预设：球坐标 (azimuth°, elevation°, distance 倍率) + 目标点
VIEW_PRESETS = {
    "front":     dict(az=0,    el=5,  dist=1.0),
    "3q4_left":  dict(az=-45,  el=15, dist=1.0),
    "3q4_right": dict(az=45,   el=15, dist=1.0),
    "side":      dict(az=90,   el=5,  dist=1.0),
    "top":       dict(az=0,    el=75, dist=1.1),
}

def ensure_dir(p):
    if not os.path.isdir(p):
        os.makedirs(p)

def set_camera_by_preset(preset, target=(0, 0, 0)):
    """按球坐标定位相机并朝向 target。KeyShot 相机 API 需按本机文档微调。"""
    import math
    az = math.radians(preset["az"])
    el = math.radians(preset["el"])
    r  = 100 * preset["dist"]        # 基准距离，按模型尺寸调整
    pos = (r * math.cos(el) * math.sin(az),
           r * math.sin(el),
           r * math.cos(el) * math.cos(az))
    lux.setCameraPosition(pos)
    lux.setCameraLookAt(target)
    lux.setCameraUp((0, 1, 0))
    lux.sync()

def main():
    for ksp, sku, views in ASSETS:
        lux.openFile(ksp)
        scene_dir = os.path.join(OUT_ROOT, sku)
        ensure_dir(scene_dir)

        # 保存原始材质，替换成中性灰白模
        # 白模做法：遍历场景对象，把材质统一成 0.8 灰的 diffuse
        original = {}
        for node in lux.getSceneTree().getChildren():
            original[node.getName()] = node.getMaterial()

        for view in views:
            set_camera_by_preset(VIEW_PRESETS[view])

            # --- Pass 1: 白模 Beauty（给 Blender 对齐 / Canny 用）---
            lux.renderImage(
                path=os.path.join(scene_dir, f"{sku}_{view}_clay.png"),
                width=W, height=H,
            )

            # --- Pass 2: 带材质 Beauty（如果需要保留真渲染作为对照）---
            for name, mat in original.items():
                if mat:
                    lux.setObjectMaterial(name, mat)
            lux.sync()
            lux.renderImage(
                path=os.path.join(scene_dir, f"{sku}_{view}_material.png"),
                width=W, height=H,
            )
            # 再切回白模，进入下一个机位
            for name in original:
                lux.setObjectMaterial(name, "Plastic_Gray")

        # 导出 GLB 给 Blender 出 depth/normal（保证几何一致）
        lux.exportFile(os.path.join(scene_dir, f"{sku}.glb"), lux.EXPORT_GLTF)

    lux.sync()

main()
```

> **实操提示**
> - `lux.renderImage()` 的 `opts` 参数可以用 `lux.getRenderOptions()` 拿到后改，比如分辨率、采样数、渲染引擎。
> - 白模材质不要真去改 KSP：**建议另存一份「白模版 KSP」**，只改材质、不动结构，避免污染原资产。
> - **最省事的做法**：直接给同一个 KSP 建多个「模型集（Model Set）」或「Studio」，白模一套、材质一套，脚本切 Studio 即可。

## 二、Blender：出 Depth / Normal / Alpha

```python
# scripts/blender_passes.py
# 用法： blender -b -P "D:/Dsektop/AI渲染/scripts/blender_passes.py" -- --glb "D:/.../LS-360G.glb" --out "D:/.../passes/LS-360G" --views front,3q4_left
#
# 更稳的替代方案：直接用 BlenderProc（pip install blenderproc），它内建 normal/depth 输出。
# 这个脚本是"不想装 BlenderProc"时的轻量版。

import bpy
import os
import sys
import math
from mathutils import Vector

argv = sys.argv[sys.argv.index("--") + 1:] if "--" in sys.argv else []

def arg(name, default=None):
    if name in argv:
        return argv[argv.index(name) + 1]
    return default

GLB = arg("--glb")
OUT = arg("--out")
VIEWS = arg("--views", "front,3q4_left,3q4_right").split(",")
RES = int(arg("--res", "2048"))

VIEW_PRESETS = {
    "front":     (0,    5,  1.0),
    "3q4_left":  (-45,  15, 1.0),
    "3q4_right": (45,   15, 1.0),
    "side":      (90,   5,  1.0),
    "top":       (0,    75, 1.1),
}

def clear_scene():
    bpy.ops.wm.read_factory_settings(use_empty=True)

def import_glb(path):
    bpy.ops.import_scene.gltf(filepath=path)
    objs = [o for o in bpy.context.scene.objects if o.type == "MESH"]
    if not objs:
        raise RuntimeError("GLB 里没有 mesh")
    # 计算整体包围盒，用于自动取景
    mins = Vector((1e9, 1e9, 1e9))
    maxs = Vector((-1e9, -1e9, -1e9))
    for o in objs:
        for c in o.bound_box:
            w = o.matrix_world @ Vector(c)
            mins = Vector((min(mins[i], w[i]) for i in range(3)))
            maxs = Vector((max(maxs[i], w[i]) for i in range(3)))
    center = (mins + maxs) / 2
    radius = (maxs - mins).length / 2
    return center, radius

def setup_render():
    scn = bpy.context.scene
    scn.render.engine = "CYCLES"          # depth/normal 通道 Cycles 最稳
    scn.cycles.samples = 32               # pass 不需要高采样
    scn.cycles.use_denoising = False
    scn.render.resolution_x = RES
    scn.render.resolution_y = RES
    scn.render.resolution_percentage = 100
    scn.render.image_settings.file_format = "PNG"
    scn.render.image_settings.color_mode = "RGB"
    scn.render.image_settings.color_depth = "16"
    scn.render.film_transparent = True    # alpha 通道
    # 开启需要的 AOV / pass
    vl = bpy.context.view_layer
    vl.use_pass_combined = True
    vl.use_pass_z = True                  # depth
    vl.use_pass_normal = True             # normal
    vl.use_pass_ao = True

def setup_compositor(out_dir, view):
    """用 Compositor 把 Z / Normal 各输出成独立文件。"""
    scn = bpy.context.scene
    scn.use_nodes = True
    tree = scn.node_tree
    tree.nodes.clear()

    rl = tree.nodes.new("CompositorNodeRLayers")

    def file_out(name, out_path, socket_index=None, socket_name=None):
        n = tree.nodes.new("CompositorNodeOutputFile")
        n.base_path = out_path
        n.file_slots[0].path = name
        n.format.file_format = "PNG"
        src = rl.outputs[socket_index] if socket_index is not None else rl.outputs[socket_name]
        tree.links.new(src, n.inputs[0])
        return n

    # 注意：Cycles 的 Z pass 是黑白浮点，ControlNet 需要归一化后的 8bit，这里直接出 EXR 更保险
    file_out("depth_",  out_dir, socket_name="Depth")
    file_out("normal_", out_dir, socket_name="Normal")
    file_out("ao_",     out_dir, socket_name="AO")
    file_out("beauty_", out_dir, socket_name="Image")

    # 主体蒙版：用 alpha
    n = tree.nodes.new("CompositorNodeOutputFile")
    n.base_path = out_dir
    n.file_slots[0].path = "alpha_"
    n.format.file_format = "PNG"
    n.format.color_mode = "BW"
    tree.links.new(rl.outputs["Alpha"], n.inputs[0])

def place_camera(center, radius, preset):
    az, el, dist = preset
    cam_data = bpy.data.cameras.new("Cam")
    cam_data.lens = 85
    cam = bpy.data.objects.new("Cam", cam_data)
    bpy.context.scene.collection.objects.link(cam)
    bpy.context.scene.camera = cam

    azr, elr = math.radians(az), math.radians(el)
    d = radius * 3.2 * dist
    loc = Vector((
        center.x + d * math.cos(elr) * math.sin(azr),
        center.y + d * math.sin(elr),
        center.z + d * math.cos(elr) * math.cos(azr),
    ))
    cam.location = loc
    direction = center - loc
    cam.rotation_euler = direction.to_track_quat("-Z", "Y").to_euler()
    return cam

def main():
    clear_scene()
    center, radius = import_glb(GLB)
    setup_render()

    for view in VIEWS:
        out_dir = os.path.join(OUT, view)
        os.makedirs(out_dir, exist_ok=True)
        cam = place_camera(center, radius, VIEW_PRESETS[view])
        setup_compositor(out_dir, view)
        bpy.context.scene.render.filepath = os.path.join(out_dir, "render_")
        bpy.ops.render.render(write_still=True)
        bpy.data.objects.remove(cam, do_unlink=True)
        print(f"[OK] {view} -> {out_dir}")

main()
```

> **重要**：`Normal` pass 是 **相机空间** 还是 **世界空间**，两个空间出来的图不一样，ControlNet 对不上会直接画出错误高光。**KeyShot 出 beauty + Blender 出 normal 时，务必确认是相机空间（camera space）法线**，并和 ControlNet 训练约定一致（通常 OpenGL 约定：X 右、Y 上、Z 朝观察者）。
>
> 出 EXR 后自己归一化，比让 Blender 直接出 8bit 更可控：
> ```python
> import cv2, numpy as np
> d = cv2.imread("depth_0001.exr", cv2.IMREAD_UNCHANGED).astype(np.float32)
> d = (d - d.min()) / (d.max() - d.min() + 1e-8)   # 近白远黑 或 反之，按 ControlNet 要求
> cv2.imwrite("depth_8bit.png", (d * 255).astype(np.uint8))
> ```

## 三、Pass 对齐校验（必做，30 行省几十小时）

```python
# scripts/align_check.py
# 生成图 vs 白模 beauty 的轮廓偏移。偏移超阈值就说明 ControlNet 没锁住，别浪费时间精修。
import cv2
import numpy as np

def silhouette(img_path, thresh=245):
    """白底图取主体轮廓。深色背景下改一下阈值策略。"""
    img = cv2.imread(img_path)
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    mask = (gray < thresh).astype(np.uint8) * 255
    # 去噪
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, np.ones((5, 5), np.uint8))
    return mask

def iou_and_offset(clay_path, gen_path):
    a = silhouette(clay_path)
    b = silhouette(gen_path)
    if a.shape != b.shape:
        b = cv2.resize(b, (a.shape[1], a.shape[0]), interpolation=cv2.INTER_NEAREST)

    inter = np.logical_and(a > 0, b > 0).sum()
    union = np.logical_or(a > 0, b > 0).sum()
    iou = inter / max(union, 1)

    # 质心偏移（归一化到图宽）
    ya, xa = np.argwhere(a > 0).mean(axis=0)
    yb, xb = np.argwhere(b > 0).mean(axis=0)
    dx = abs(xa - xb) / a.shape[1]
    dy = abs(ya - yb) / a.shape[0]

    return iou, dx, dy

if __name__ == "__main__":
    import sys
    IoU_MIN, OFFSET_MAX = 0.90, 0.02   # 经验阈值，按你的类目微调
    ok = True
    for pair in sys.argv[1:]:
        clay, gen = pair.split(",")
        iou, dx, dy = iou_and_offset(clay, gen)
        verdict = "PASS" if (iou >= IoU_MIN and max(dx, dy) <= OFFSET_MAX) else "FAIL"
        if verdict == "FAIL":
            ok = False
        print(f"{verdict}  IoU={iou:.3f}  dx={dx:.4f}  dy={dy:.4f}  {gen}")
    sys.exit(0 if ok else 1)
```

## 四、ComfyUI 批量客户端

```python
# scripts/comfy_batch.py
# 依赖： requests
# 前置： ComfyUI 以常驻服务启动
#        python main.py --listen 127.0.0.1 --port 8188 --output "D:/Dsektop/AI渲染/output/raw"
# 工作流： 在 UI 里开 Dev Mode → Save (API format) → workflows/product_render_api.json

import json
import os
import sqlite3
import time
import uuid
import copy
import requests

COMFY = "http://127.0.0.1:8188"
WORKFLOW_PATH = r"D:\Dsektop\AI渲染\workflows\product_render_api.json"
DB_PATH = r"D:\Dsektop\AI渲染\config\tasks.db"

# 工作流里可参数化的节点位置（node_id, "inputs", 字段名）
SLOTS = {
    "positive":   ("6", "inputs", "text"),
    "negative":   ("7", "inputs", "text"),
    "seed":       ("3", "inputs", "seed"),
    "width":      ("5", "inputs", "width"),
    "height":     ("5", "inputs", "height"),
    "depth_img":  ("10", "inputs", "image"),      # LoadImage（depth）
    "normal_img": ("11", "inputs", "image"),      # LoadImage（normal）
    "ref_img":    ("12", "inputs", "image"),      # LoadImage（参考图）
    "depth_w":    ("20", "inputs", "strength"),   # ControlNet Apply
    "normal_w":   ("21", "inputs", "strength"),
    "redux_f":    ("30", "inputs", "downsampling_factor"),
    "denoise":    ("3", "inputs", "denoise"),
}

# ---------------- 队列 ----------------

def init_db():
    con = sqlite3.connect(DB_PATH)
    con.execute("""
        CREATE TABLE IF NOT EXISTS tasks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            sku TEXT, view TEXT, variant INTEGER,
            card TEXT, workflow TEXT,
            status TEXT DEFAULT 'pending',   -- pending/running/done/failed
            prompt_id TEXT, output TEXT,
            retry INTEGER DEFAULT 0,
            err TEXT,
            created_at TEXT DEFAULT CURRENT_TIMESTAMP,
            updated_at TEXT DEFAULT CURRENT_TIMESTAMP
        )
    """)
    con.commit()
    return con

def claim_task(con, n=1):
    """原子取任务。断点续跑：进程挂了重跑，running 的会被重置。"""
    con.execute("UPDATE tasks SET status='pending' WHERE status='running'")
    rows = con.execute(
        "SELECT id, sku, view, variant, card, workflow FROM tasks "
        "WHERE status='pending' ORDER BY id LIMIT ?", (n,)
    ).fetchall()
    for r in rows:
        con.execute("UPDATE tasks SET status='running', updated_at=CURRENT_TIMESTAMP WHERE id=?", (r[0],))
    con.commit()
    return rows

def finish_task(con, tid, status, prompt_id=None, output=None, err=None):
    con.execute(
        "UPDATE tasks SET status=?, prompt_id=?, output=?, err=?, "
        "updated_at=CURRENT_TIMESTAMP WHERE id=?",
        (status, prompt_id, output, err, tid)
    )
    con.commit()

# ---------------- ComfyUI ----------------

def load_workflow():
    with open(WORKFLOW_PATH, "r", encoding="utf-8") as f:
        return json.load(f)

def set_slot(wf, key, value):
    if key not in SLOTS:
        raise KeyError(f"未定义的参数槽：{key}")
    node_id, _, field = SLOTS[key]
    wf[node_id]["inputs"][field] = value

def upload_image(local_path):
    """把本地图片上传到 ComfyUI，返回可用于 LoadImage 的文件名。"""
    name = os.path.basename(local_path)
    with open(local_path, "rb") as f:
        r = requests.post(f"{COMFY}/upload/image",
                          files={"image": (name, f, "image/png")},
                          data={"overwrite": "true"}, timeout=60)
    r.raise_for_status()
    return r.json()["name"]

def queue(wf):
    payload = {"prompt": wf, "client_id": str(uuid.uuid4())}
    r = requests.post(f"{COMFY}/prompt", json=payload, timeout=60)
    if r.status_code != 200:
        raise RuntimeError(f"提交失败 {r.status_code}: {r.text[:500]}")
    return r.json()["prompt_id"]

def wait(prompt_id, timeout=600, poll=1.5):
    """/history 只返回已完成任务；在途任务不出现。第一次查不到 ≠ 失败。"""
    t0 = time.time()
    while time.time() - t0 < timeout:
        r = requests.get(f"{COMFY}/history/{prompt_id}", timeout=30)
        hist = r.json()
        if prompt_id in hist:
            outputs = hist[prompt_id].get("outputs", {})
            images = []
            for node_out in outputs.values():
                for img in node_out.get("images", []):
                    images.append(img)
            if images:
                return images
            return []
        time.sleep(poll)
    raise TimeoutError(f"{prompt_id} 超时（{timeout}s）。检查 ComfyUI 终端是否有 CUDA 报错。")

def download(img_info, save_dir):
    os.makedirs(save_dir, exist_ok=True)
    r = requests.get(f"{COMFY}/view",
                     params={k: img_info[k] for k in ("filename", "subfolder", "type") if k in img_info},
                     timeout=120)
    r.raise_for_status()
    path = os.path.join(save_dir, img_info["filename"])
    with open(path, "wb") as f:
        f.write(r.content)
    return path

# ---------------- 主循环 ----------------

def build_workflow(base, card, view, variant):
    wf = copy.deepcopy(base)
    mat = card["material"]
    lit = card["lighting"]
    bg  = card["background"]
    ref = card.get("reference", {})

    set_slot(wf, "positive", build_prompt(card))
    set_slot(wf, "negative", NEGATIVE)
    set_slot(wf, "seed", card["_seed_base"] + variant)
    set_slot(wf, "width", 1024)
    set_slot(wf, "height", 1024)
    set_slot(wf, "depth_img",  upload_image(f"{card['_pass_dir']}/{view}/depth.png"))
    set_slot(wf, "normal_img", upload_image(f"{card['_pass_dir']}/{view}/normal.png"))
    if ref.get("image"):
        set_slot(wf, "ref_img", upload_image(ref["image"]))
    set_slot(wf, "depth_w",  card.get("_depth_w", 0.65))
    set_slot(wf, "normal_w", card.get("_normal_w", 0.40))
    set_slot(wf, "redux_f",  {"L1_style": 3, "L2_material": 2, "L3_full": 1}
                            .get(ref.get("strength", "L1_style"), 3))
    return wf

MATERIAL_EN = {
    "ABS": "ABS plastic", "PC": "polycarbonate", "aluminum": "aluminum",
    "PMMA": "acrylic", "rubber": "TPE rubber",
}
FINISH_EN = {
    "fine_sandblast": "fine sandblasted matte", "anodized_brushed": "anodized brushed",
    "high_gloss": "high gloss", "soft_touch": "soft-touch coating",
}

NEGATIVE = ("blurry, low quality, distorted shape, warped geometry, extra parts, "
            "text, watermark, letters, numbers, logo, label, ruler markings, "
            "reflections of other objects, people, hands, cluttered background, "
            "oversaturated, cartoon, illustration, CGI plastic look")

def build_prompt(card):
    m = card["material"]
    l = card["lighting"]
    b = card["background"]
    c = card["camera"]

    body = m["body"]
    body_desc = f"{FINISH_EN.get(body['finish'], body['finish'])} {MATERIAL_EN.get(body['type'], body['type'])} body in {body['color']}"

    parts = []
    if m.get("trim"):
        t = m["trim"]
        parts.append(f"{FINISH_EN.get(t['finish'], t['finish'])} {MATERIAL_EN.get(t['type'], t['type'])} trim in {t['color']}")
    if m.get("window"):
        parts.append("clear acrylic transparent window with realistic refraction")

    lighting_map = {
        "studio_softbox_3point": "professional studio three-point softbox lighting, soft large key light from front-left, soft fill from front-right, subtle rim light, soft contact shadow",
        "top_soft": "soft overhead diffused lighting, even illumination, minimal shadow",
        "backlight_dramatic": "dramatic backlight with soft front fill, bright edge separation",
        "natural_window": "natural window light from the side, soft gradient falloff",
    }
    bg_map = {
        "gradient_gray": "seamless light gray gradient background",
        "pure_white": "seamless pure white background",
        "scene_desktop": "minimal wooden desktop workspace scene, shallow depth of field",
    }

    prompt = (
        f"Professional commercial product photograph of a {card['asset']['category']} (do not change its shape), "
        + body_desc + (", " + ", ".join(parts) if parts else "") + ". "
        + lighting_map.get(l["scheme"], l["scheme"]) + ". "
        + bg_map.get(b["type"], b["type"]) + ". "
        + f"Shot on {c['focal']}mm lens, {VIEW_PHRASE.get(card['_current_view'], 'front view')}, "
          "sharp focus, high detail, photorealistic, 8k product photography, "
          "accurate proportions, clean surfaces, studio quality"
    )
    return prompt

VIEW_PHRASE = {
    "front": "straight-on front view",
    "3q4_left": "front three-quarter view from the left",
    "3q4_right": "front three-quarter view from the right",
    "side": "side profile view",
    "top": "top-down view",
    "detail_keypad": "close-up macro detail view",
}

def main():
    con = init_db()
    base = load_workflow()
    stats = {"ok": 0, "fail": 0}

    while True:
        rows = claim_task(con, n=1)
        if not rows:
            break
        tid, sku, view, variant, card_json, _ = rows[0]
        card = json.loads(card_json)
        card["_current_view"] = view
        try:
            wf = build_workflow(base, card, view, variant)
            pid = queue(wf)
            images = wait(pid, timeout=card.get("_timeout", 600))
            saved = [download(i, os.path.join(r"D:\Dsektop\AI渲染\output\raw", sku)) for i in images]
            finish_task(con, tid, "done", pid, ";".join(saved))
            stats["ok"] += 1
            print(f"[OK] {sku}/{view}/v{variant} -> {saved}")
        except Exception as e:
            retry = con.execute("SELECT retry FROM tasks WHERE id=?", (tid,)).fetchone()[0]
            if retry < 2:
                con.execute("UPDATE tasks SET status='pending', retry=retry+1 WHERE id=?", (tid,))
                con.commit()
                print(f"[RETRY {retry+1}] {sku}/{view}/v{variant}: {e}")
            else:
                finish_task(con, tid, "failed", err=str(e))
                stats["fail"] += 1
                print(f"[FAIL] {sku}/{view}/v{variant}: {e}")

    print(f"\n完成：成功 {stats['ok']}，失败 {stats['fail']}")

if __name__ == "__main__":
    main()
```

> **踩坑对照**（对应方案第八节的表格）

## 六、PS 脚本：LOGO / 丝印 / 刻度回贴

```javascript
// scripts/logo_relight.jsx  —  放进 Photoshop 的 Presets/Scripts 或直接 File > Scripts > Browse
// 作用：把 LOGO 图层按下方渲染图的光影做正片叠底/叠加，实现"穿在材质上"的效果
// 用法：先打开渲染图，再把这个脚本跑起来，它会找名为 "LOGO_SRC" 的智能对象图层

#target photoshop

app.brand = false;
app.displayDialogs = DialogModes.NO;

function main() {
    var doc = app.activeDocument;
    var logo = doc.layers.getByName("LOGO_SRC");

    // 1. 复制一份作为基底
    var base = logo.duplicate();
    base.name = "LOGO_BASE";

    // 2. 复制一份做叠加层（取光影）
    var shadow = logo.duplicate();
    shadow.name = "LOGO_SHADOW";
    shadow.blendMode = BlendMode.MULTIPLY;
    shadow.opacity = 55;

    // 3. 高光层
    var hi = logo.duplicate();
    hi.name = "LOGO_HIGHLIGHT";
    hi.blendMode = BlendMode.SCREEN;
    hi.opacity = 25;

    // 4. 全部编组，方便一次性调
    var group = doc.layerSets.add();
    group.name = "LOGO_GROUP";
    base.move(group, ElementPlacement.INSIDE);
    shadow.move(group, ElementPlacement.INSIDE);
    hi.move(group, ElementPlacement.INSIDE);

    alert("LOGO 回贴完成。请检查：\n" +
          "- 不透明度是否需按材质调整（磨砂 55%、高光件 40%、金属 65%）\n" +
          "- 透视是否与产品面贴合");
}

main();
```

> 更稳的方案：把 LOGO / 刻度做成**白底黑字的蒙版 PNG**，在渲染图上用「置换 + 叠加」模式贴。批量时用这个脚本跑，比逐张手工快 50 倍以上。

## 八、参数调优速查（贴墙版）

| 现象 | 原因 | 调法 |
|---|---|---|
| 轮廓不对 / 厚度关系错 | Depth 权重太低 | depth 0.60 → 0.75 |
| 画面发灰、死塑料感 | Normal 权重过高 或 Depth 过高 | normal 0.50 → 0.35；depth 降 0.05 |
| 高光走向不合理（金属拉丝方向错） | Normal 空间搞错 | 确认是**相机空间**法线，与训练约定一致 |
| 参考图只复刻、不听 prompt | Redux 权重太高 | `downsampling_factor` 1 → 3，或降 `weight` |
| 参考图影响太弱、感觉没用 | Redux 权重太低 | `downsampling_factor` 3 → 1 |
| 背景颜色渗到产品上 | 用了 IC-Light `fbc` | 换 `fc`，背景写进 prompt |
| 刻度 / 文字糊掉 | prompt 里没排除 | 负向词加 `text, numbers, markings`；LOGO 走回贴 |
| 透明件崩 | AI 对透明材质能力不足 | 降低 AI 占比，透明区域用 KeyShot 真渲染合成 |
| 同 SKU 多图不一致 | seed 与参考图漂移 | 固定 seed 基数 + 同一参考图 + prompt 骨架只换风格词 |
| 1536 分辨率爆显存 | ControlNet 数量太多 | 1536 最多叠 2 个；或 1024 出图后超分 |
| 队列卡住不报错 | 节点静默失败 | 读 ComfyUI 终端日志，必设 timeout |
